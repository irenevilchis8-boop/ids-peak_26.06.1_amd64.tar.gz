50class GenericAutoFeatures : public IAutoFeature
51{
52 static constexpr int moduleVersion = 1;
53
54public:
55 friend class autofeature::BrightnessComponent;
56
57 explicit GenericAutoFeatures(const std::shared_ptr<peak::core::Device>& device, bool advanced = true)
58 : m_remoteDeviceNodeMap{ device ?
59 device->RemoteDevice()->NodeMaps().at(0) :
60 throw peak::afl::error::InvalidParameterException("The given device is not valid!", PEAK_AFL_STATUS_ERROR) }
61 , m_manager{ m_remoteDeviceNodeMap }
62 {
63 const auto isColorCamera = [this]() {
64 const auto pixelFormatNode = m_remoteDeviceNodeMap->TryFindNode<peak::core::nodes::EnumerationNode>("PixelFormat");
65 if (pixelFormatNode == nullptr || !pixelFormatNode->IsReadable())
66 {
67 return false;
68 }
69
70 for (const auto& entry : pixelFormatNode->AvailableEntries())
71 {
72 try
73 {
74 const auto pixelFormatInfo = peak::common::PixelFormatInfo(static_cast<peak::common::PixelFormat>(entry->Value()));
75 // If the camera supports bayer it is a color camera
76 if (pixelFormatInfo.HasChannel(peak::common::Channel::Bayer) || !pixelFormatInfo.IsSingleChannel())
77 {
78 return true;
79 }
80 }
81 catch (const std::exception&)
82 {
83 // ignore unknown pixelformats
84 }
85 }
86
87 return false;
88 };
89
90 for (const auto& controllerType :
91 { autofeature::ControllerType::Focus, autofeature::ControllerType::Brightness, autofeature::ControllerType::WhiteBalance })
92 {
93 if (controllerType == autofeature::ControllerType::WhiteBalance && !isColorCamera())
94 {
95 continue;
96 }
97
98 if (m_manager.IsControllerSupported(autofeature::detail::ToCType(controllerType)))
99 {
100 auto controller = CreateController(m_manager, controllerType, advanced);
101 m_controllers.emplace_back(std::move(controller));
102 }
103 }
104
105 m_manager.SetGainIPL(m_gain);
106
107 GenericAutoFeatures::ResetToDefault();
108 }

137 {
138 const auto version{ archive.GetInt("Version") };
139 if (version < 1)

143 }
144
145 m_enabled = archive.GetBool("Enabled");
146 for (const auto& controller : m_controllers)
147 {
148 auto controllerArchive = archive.GetArchive(controller->GetName());
149 controller->Deserialize(*controllerArchive);
150 }
151 }

237private:
238 PEAK_COMMON_NO_DISCARD std::unique_ptr<autofeature::detail::IController> CreateController(
239 peak::afl::Manager& manager, autofeature::ControllerType controllerType, bool advanced)
240 {
241 switch (controllerType)
242 {
243 case autofeature::ControllerType::Focus:
244 if (advanced)
245 {
246 return std::make_unique<autofeature::AdvancedAutoFocus>(manager);
247 }
248 return std::make_unique<autofeature::BasicAutoFocus>(manager);
249
250 case autofeature::ControllerType::Brightness: {
251 auto callback = [this]() {
252 if (!m_gainModule)
253 {
254 return;
255 }
256 m_gainModule->SetMaster(m_gain.MasterGainValue());
257 };
258
259 if (advanced)
260 {
261 return std::make_unique<autofeature::AdvancedAutoBrightness>(manager, callback);
262 }
263 return std::make_unique<autofeature::BasicAutoBrightness>(manager, callback);
264 }
265
266 case autofeature::ControllerType::WhiteBalance:
267 auto callback = [this]() {
268 if (!m_gainModule)
269 {
270 return;
271 }
272
273 m_gainModule->SetRed(m_gain.RedGainValue());
274 m_gainModule->SetGreen(m_gain.GreenGainValue());
275 m_gainModule->SetBlue(m_gain.BlueGainValue());
276 };
277
278 if (advanced)
279 {
280 return std::make_unique<autofeature::AdvancedAutoWhiteBalance>(manager, callback);
281 }
282 return std::make_unique<autofeature::BasicAutoWhiteBalance>(manager, callback);
283 }
284 throw peak::afl::error::InternalErrorException("Unknown controller type", PEAK_AFL_STATUS_ERROR);
285 }

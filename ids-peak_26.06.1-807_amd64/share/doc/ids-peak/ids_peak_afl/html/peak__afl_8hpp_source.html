567static void ThrowException(peak_afl_status code, bool get_text = true)
568{
569 const auto text = get_text ? library::GetLastError() : "No additional text";
570
571 switch (code)
572 {
573 case PEAK_AFL_STATUS_ERROR:
574 throw InternalErrorException(text, code);
575
576 case PEAK_AFL_STATUS_NOT_INITIALIZED:
577 throw NotInitializedException(text, code);
578
579 case PEAK_AFL_STATUS_INVALID_PARAMETER:
580 throw InvalidParameterException(text, code);
581
582 case PEAK_AFL_STATUS_ACCESS_DENIED:
583 throw BadAccessException(text, code);
584
585 case PEAK_AFL_STATUS_BUSY:
586 throw BusyException(text, code);
587
588 case PEAK_AFL_STATUS_BUFFER_TOO_SMALL:
589 throw BufferTooSmallException(text, code);
590
591 case PEAK_AFL_STATUS_INVALID_IMAGE_FORMAT:
592 throw InvalidImageFormatException(text, code);
593
594 case PEAK_AFL_STATUS_NOT_SUPPORTED:
595 throw NotSupportedException(text, code);
596
597 case PEAK_AFL_STATUS_VALUE_ADJUSTED:
598 case PEAK_AFL_STATUS_SUCCESS:
599 break;
600 }

871template <typename... ControllerTypes>
872std::tuple<Manager, std::vector<std::shared_ptr<Controller>>> CreateManager(
873 const std::shared_ptr<peak::core::NodeMap>& node_map, ControllerTypes... types)
874{
875 constexpr int size = sizeof...(types);
876 constexpr std::array<peak_afl_controllerType, size> controller_types{ types... };
877 std::vector<std::shared_ptr<Controller>> controller{};
878 Manager manager{ node_map };
879 controller.reserve(size);
880
881 std::for_each(controller_types.cbegin(), controller_types.cend(),
882 [&manager, &controller](peak_afl_controllerType type) { controller.emplace_back(manager.CreateController(type)); });
883
884 return { std::move(manager), std::move(controller) };

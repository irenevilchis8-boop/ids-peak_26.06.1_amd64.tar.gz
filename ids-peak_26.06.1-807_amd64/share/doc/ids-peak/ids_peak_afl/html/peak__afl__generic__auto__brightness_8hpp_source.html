44inline std::unique_ptr<features::BrightnessLimit> CreateBrightnessLimit(
45 std::shared_ptr<peak::afl::Controller> controller, BrightnessComponentType type)
46{
47 switch (type)
48 {
49 case BrightnessComponentType::Exposure:
50 return std::make_unique<features::BrightnessExposureLimit>(std::move(controller));
51 case BrightnessComponentType::Gain:
52 return std::make_unique<features::BrightnessGainLimit>(std::move(controller));
53 case BrightnessComponentType::AnalogGain:
54 return std::make_unique<features::BrightnessGainAnalogLimit>(std::move(controller));
55 case BrightnessComponentType::DigitalGain:
56 return std::make_unique<features::BrightnessGainDigitalLimit>(std::move(controller));
57 case BrightnessComponentType::CombinedGain:
58 return std::make_unique<features::BrightnessGainCombinedLimit>(std::move(controller));
59 case BrightnessComponentType::HostGain:
60 return std::make_unique<features::BrightnessGainHostLimit>(std::move(controller));
61 default:
62 throw peak::afl::error::InvalidParameterException("Unknown brightness component type", PEAK_AFL_STATUS_INVALID_PARAMETER);
63 }
64}

73 explicit GenericAutoBrightness(
74 peak::afl::Manager& manager, ProcessingCallback callback, const std::vector<BrightnessComponentType>& componentTypes)
75 : IController(manager)
76 , m_controller{ peak::afl::Controller::Create(PEAK_AFL_CONTROLLER_TYPE_BRIGHTNESS) }
77 , m_processingCallback{ std::move(callback) }
78 , m_skipFrames{ std::make_unique<features::SkipFrames>(m_controller) }
79 , m_autoTarget{ std::make_unique<features::AutoTarget>(m_controller) }
80 , m_autoTolerance{ std::make_unique<features::AutoTolerance>(m_controller) }
81 , m_autoPercentile{ std::make_unique<features::AutoPercentile>(m_controller) }
82 , m_roi{ std::make_unique<features::Roi>(m_controller) }
83 , m_brightnessAlgorithm{ std::make_unique<features::BrightnessAlgorithm>(m_controller) }
84 {
85 GetManager().AddController(m_controller);
86
87 m_brightnessComponents.reserve(componentTypes.size());
88 for (const auto& componentType : componentTypes)
89 {
90 if (!m_controller->IsBrightnessComponentUnitSupported(detail::ToCType(componentType)))
91 {
92 continue;
93 }
94
95 m_brightnessComponents.emplace_back(std::unique_ptr<BrightnessComponent>(
96 new BrightnessComponent(m_controller, componentType, detail::CreateBrightnessLimit(m_controller, componentType))));
97 }
98
99 m_controller->RegisterFinishedCallback([this]() { m_FinishedCallbackManager.TriggerCallbacks(); });
100
101 m_controller->RegisterProcessingCallback([this](peak_afl_process_data* data) {
102 auto* brightness = reinterpret_cast<peak_afl_process_data_brightness*>(data);
103 if (brightness->controller_component != PEAK_AFL_CONTROLLER_BRIGHTNESS_COMPONENT_HOST_GAIN)
104 {
105 return;
106 }
107
108 if (m_processingCallback)
109 {
110 m_processingCallback();
111 }
112 });
113 }

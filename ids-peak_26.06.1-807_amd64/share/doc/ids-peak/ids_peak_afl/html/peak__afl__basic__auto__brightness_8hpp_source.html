70PEAK_COMMON_NO_DISCARD inline std::string ToString(AutoBrightnessPolicy policy)
71{
72 switch (policy)
73 {
74 case AutoBrightnessPolicy::ExposureAndGain:
75 return "ExposureAndGain";
76 case AutoBrightnessPolicy::ExposureOnly:
77 return "ExposureOnly";
78 case AutoBrightnessPolicy::GainOnly:
79 return "GainOnly";
80 }
81
82 throw peak::afl::error::InvalidParameterException("The given policy is unknown!", PEAK_AFL_STATUS_INVALID_PARAMETER);
83}

93PEAK_COMMON_NO_DISCARD inline AutoBrightnessPolicy ToAutoBrightnessPolicy(const std::string& name)

201 void SetMode(ControllerMode mode) override
202 {
203 auto* exposureComponent = GetExposureComponent();
204 if (exposureComponent != nullptr)
205 {
206 const auto controllerMode = (m_policy == AutoBrightnessPolicy::ExposureOnly
207 || m_policy == AutoBrightnessPolicy::ExposureAndGain) ?
208 mode :
209 ControllerMode::Off;
210
211 exposureComponent->SetMode(controllerMode);
212 }
213
214 auto* gainComponent = GetGainComponent();
215 if (gainComponent != nullptr)
216 {
217 const auto controllerMode = (m_policy == AutoBrightnessPolicy::GainOnly || m_policy == AutoBrightnessPolicy::ExposureAndGain) ?
218 mode :
219 ControllerMode::Off;
220
221 gainComponent->SetMode(controllerMode);
222 }
223
224 m_cachedMode = mode;
225 }

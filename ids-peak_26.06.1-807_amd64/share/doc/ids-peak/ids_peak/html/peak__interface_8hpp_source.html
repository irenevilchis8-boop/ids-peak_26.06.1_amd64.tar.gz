608inline std::shared_ptr<DeviceDescriptor> Interface::GetOrAddFoundDevice(PEAK_DEVICE_DESCRIPTOR_HANDLE foundDevice)
609{
610 std::lock_guard<std::mutex> lock(m_devicesMutex);
611
612 const auto deviceKey = QueryStringFromCInterfaceFunction([&](char* key, size_t* keySize) {
613 return PEAK_C_ABI_PREFIX PEAK_DeviceDescriptor_GetKey(foundDevice, key, keySize);
614 });
615 auto keyDevicePairIterator = m_devicesByKey.find(deviceKey);
616 if (keyDevicePairIterator == m_devicesByKey.end())
617 {
618 auto deviceDescriptor = std::static_pointer_cast<DeviceDescriptor>(
619 std::make_shared<ClassCreator<DeviceDescriptor>>(
620 foundDevice, std::weak_ptr<Interface>{ shared_from_this() }));
621 m_devices.emplace_back(deviceDescriptor);
622 m_devicesByKey.emplace(deviceKey, deviceDescriptor);
623 m_devicesKeyById.emplace(deviceDescriptor->ID(), deviceKey);
624
625 return deviceDescriptor;
626 }
627
628 return keyDevicePairIterator->second;
629}

652inline void Interface::InitializeUpdateMechanismIfNecessary()
653{
654 std::call_once(m_updateMechanismInitializedFlag, [&] {
655 (void)RegisterDeviceFoundCallback([](const std::shared_ptr<DeviceDescriptor>&) {
656 // Registering an empty callback is enough since DeviceFoundCallbackCWrapper() handles everything else.
657 });
658
659 (void)RegisterDeviceLostCallback([&](const std::string& lostDeviceId) {
660 {
661 std::lock_guard<std::mutex> lock(m_devicesMutex);
662 const auto lostDeviceKey = m_devicesKeyById.at(lostDeviceId);
663 m_devices.erase(std::remove_if(std::begin(m_devices), std::end(m_devices),
664 [lostDeviceKey](const std::shared_ptr<DeviceDescriptor>& device) {
665 return device->Key() == lostDeviceKey;
666 }),
667 std::end(m_devices));
668 m_devicesByKey.erase(lostDeviceKey);
669 m_devicesKeyById.erase(lostDeviceId);
670 }
671 });
672 });

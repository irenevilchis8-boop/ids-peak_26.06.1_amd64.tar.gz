112 return PEAK_C_ABI_PREFIX PEAK_Module_GetNumNodeMaps(moduleHandle, _numNodeMaps);

114

119 [&](PEAK_NODE_MAP_HANDLE* _nodeMapHandle) {
120 return PEAK_C_ABI_PREFIX PEAK_Module_GetNodeMap(moduleHandle, x, _nodeMapHandle);

124 }

129 return PEAK_C_ABI_PREFIX PEAK_Module_GetPort(moduleHandle, _portHandle);
130 });
131
132 m_port = std::make_shared<ClassCreator<peak::core::Port>>(portHandle);

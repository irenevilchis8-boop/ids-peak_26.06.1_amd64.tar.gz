309inline std::vector<std::string> FileAdapter::AvailableFileNames(const std::shared_ptr<NodeMap>& nodeMap)
310{
311 if (nodeMap == nullptr)
312 {
313 throw InvalidArgumentException("The given nodeMap is a nullptr!");
314 }
315
316 const auto fileSelectorNode = nodeMap->TryFindNode<nodes::EnumerationNode>("FileSelector");
317 if (fileSelectorNode == nullptr)
318 {
319 throw NotAvailableException("EnumerationNode with the name 'FileSelector' not found.");
320 }
321 std::vector<std::string> availableFileNames{};
322 const auto entries = fileSelectorNode->AvailableEntries();
323 availableFileNames.reserve(entries.size());
324
325 for (const auto& entry : entries)
326 {
327 availableFileNames.emplace_back(entry->SymbolicValue());
328 }
329 return availableFileNames;
330}

363inline void Port::Initialize() const
364{
365 auto numUrls = QueryNumericFromCInterfaceFunction<size_t>(
366 [&](size_t* _numUrls) { return PEAK_C_ABI_PREFIX PEAK_Port_GetNumURLs(m_backendHandle, _numUrls); });
367
368 std::vector<std::shared_ptr<PortURL>> urls;
369 for (size_t x = 0; x < numUrls; ++x)
370 {
371 auto portUrlHandle = QueryNumericFromCInterfaceFunction<PEAK_PORT_URL_HANDLE>(
372 [&](PEAK_PORT_URL_HANDLE* _portUrlHandle) {
373 return PEAK_C_ABI_PREFIX PEAK_Port_GetURL(m_backendHandle, x, _portUrlHandle);
374 });
375
376 urls.emplace_back(
377 std::make_shared<ClassCreator<PortURL>>(portUrlHandle, std::const_pointer_cast<Port>(shared_from_this())));
378 }
379
380 m_portURLs = urls;
381}

746inline void PEAK_CALL_CONV DeviceDescriptor::InformationChangedCallbackCWrapper(
747 const PEAK_DEVICE_INFORMATION_ROLE* changedRoles, size_t changedRolesSize, void* context)
748{
749 auto callback = static_cast<DeviceDescriptor::InformationChangedCallback*>(context);
750 std::vector<DeviceInformationRole> changedRolesVec;
751 changedRolesVec.reserve(changedRolesSize);
752 for (size_t i = 0; i < changedRolesSize; ++i)
753 {
754 changedRolesVec.emplace_back(static_cast<DeviceInformationRole>(*(changedRoles + i)));
755 }
756 callback->operator()(changedRolesVec);
757}

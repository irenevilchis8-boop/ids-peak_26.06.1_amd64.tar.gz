380inline std::vector<std::shared_ptr<EnumerationEntryNode>> EnumerationNode::Entries() const
381{
382 auto numEntries = QueryNumericFromCInterfaceFunction<size_t>([&](size_t* _numEntries) {
383 return PEAK_C_ABI_PREFIX PEAK_EnumerationNode_GetNumEntries(m_backendHandle, _numEntries);
384 });
385
386 std::vector<std::shared_ptr<EnumerationEntryNode>> entries;
387 for (size_t x = 0; x < numEntries; ++x)
388 {
389 auto enumerationEntryNodeHandle = QueryNumericFromCInterfaceFunction<PEAK_ENUMERATION_ENTRY_NODE_HANDLE>(
390 [&](PEAK_ENUMERATION_ENTRY_NODE_HANDLE* _enumerationEntryNodeHandle) {
391 return PEAK_C_ABI_PREFIX PEAK_EnumerationNode_GetEntry(
392 m_backendHandle, x, _enumerationEntryNodeHandle);
393 });
394
395 auto nodeHandle = QueryNumericFromCInterfaceFunction<PEAK_NODE_HANDLE>(
396 [&](PEAK_NODE_HANDLE* _nodeHandle) {
397 return PEAK_C_ABI_PREFIX PEAK_EnumerationEntryNode_ToNode(
398 enumerationEntryNodeHandle, _nodeHandle);
399 });
400
401 auto nodeName = GetNodeName(nodeHandle);
402
403 entries.emplace_back(std::dynamic_pointer_cast<EnumerationEntryNode>(ParentNodeMap()->FindNode(nodeName)));
404 }
405
406 return entries;
407}

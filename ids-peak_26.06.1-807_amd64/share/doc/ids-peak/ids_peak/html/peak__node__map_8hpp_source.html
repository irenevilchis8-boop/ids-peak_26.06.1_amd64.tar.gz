686inline void NodeMap::Initialize() const
687{
688 auto numNodes = QueryNumericFromCInterfaceFunction<size_t>([&](size_t* _numNodes) {
689 return PEAK_C_ABI_PREFIX PEAK_NodeMap_GetNumNodes(m_backendHandle, _numNodes);
690 });
691
692 for (size_t x = 0; x < numNodes; ++x)
693 {
694 auto nodeHandle = QueryNumericFromCInterfaceFunction<PEAK_NODE_HANDLE>(
695 [&](PEAK_NODE_HANDLE* _nodeHandle) {
696 return PEAK_C_ABI_PREFIX PEAK_NodeMap_GetNode(m_backendHandle, x, _nodeHandle);
697 });
698
699 auto node = NodeFromHandle(nodeHandle);
700
701 m_nodes.emplace_back(node);
702 m_nodesByName.emplace(node->Name(), node);
703 }
704}

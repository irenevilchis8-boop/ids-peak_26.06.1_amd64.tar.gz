895inline std::shared_ptr<Buffer> DataStream::AnnounceBuffer(
896 void* buffer, size_t size, void* userPtr, const DataStream::BufferRevocationCallback& revocationCallback)
897{
898 auto _callback = std::make_unique<BufferRevocationCallback>(revocationCallback);
899
900 auto callbackContext = _callback.get();
901
902 auto bufferHandle = QueryNumericFromCInterfaceFunction<PEAK_BUFFER_HANDLE>(
903 [&](PEAK_BUFFER_HANDLE* _bufferHandle) {
904 return PEAK_C_ABI_PREFIX PEAK_DataStream_AnnounceBuffer(m_backendHandle, buffer, size, userPtr,
905 BufferRevocationCallbackCWrapper, callbackContext, _bufferHandle);
906 });
907
908 auto _buffer = std::make_shared<ClassCreator<Buffer>>(bufferHandle, shared_from_this());
909 AddAnnouncedBuffer(_buffer);
910
911 {
912 std::lock_guard<std::mutex> lock(m_bufferRevocationCallbacksByBufferHandleMutex);
913
914 m_bufferRevocationCallbacksByBufferHandle.emplace(bufferHandle, std::move(_callback));
915 }
916
917 return _buffer;
918}

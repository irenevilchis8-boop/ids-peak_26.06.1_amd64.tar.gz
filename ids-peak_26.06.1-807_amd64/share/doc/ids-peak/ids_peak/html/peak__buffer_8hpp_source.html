612inline std::string ToString(BufferPayloadType entry)
613{
614 std::string entryString;
615
616 if (entry == BufferPayloadType::Unknown)
617 {
618 entryString = "Unknown";
619 }
620 else if (entry == BufferPayloadType::Image)
621 {
622 entryString = "Image";
623 }
624 else if (entry == BufferPayloadType::RawData)
625 {
626 entryString = "RawData";
627 }
628 else if (entry == BufferPayloadType::File)
629 {
630 entryString = "File";
631 }
632 else if (entry == BufferPayloadType::Chunk)
633 {
634 entryString = "Chunk";
635 }
636 else if (entry == BufferPayloadType::JPEG)
637 {
638 entryString = "JPEG";
639 }
640 else if (entry == BufferPayloadType::JPEG2000)
641 {
642 entryString = "JPEG2000";
643 }
644 else if (entry == BufferPayloadType::H264)
645 {
646 entryString = "H264";
647 }
648 else if (entry == BufferPayloadType::ChunkOnly)
649 {
650 entryString = "ChunkOnly";
651 }
652 else if (entry == BufferPayloadType::DeviceSpecific)
653 {
654 entryString = "DeviceSpecific";

656 else if (entry == BufferPayloadType::MultiPart)

658 entryString = "MultiPart";

940{

942
943 auto numChunks = QueryNumericFromCInterfaceFunction<size_t>([&](size_t* _numChunks) {
944 return PEAK_C_ABI_PREFIX PEAK_Buffer_GetNumChunks(m_backendHandle, _numChunks);
945 });
946
947 std::vector<std::shared_ptr<BufferChunk>> chunks;
948 for (size_t x = 0; x < numChunks; ++x)
949 {
950 auto bufferChunkHandle = QueryNumericFromCInterfaceFunction<PEAK_BUFFER_CHUNK_HANDLE>(
951 [&](PEAK_BUFFER_CHUNK_HANDLE* _bufferChunkHandle) {
952 return PEAK_C_ABI_PREFIX PEAK_Buffer_GetChunk(m_backendHandle, x, _bufferChunkHandle);
953 });
954
955 chunks.emplace_back(std::make_shared<ClassCreator<BufferChunk>>(bufferChunkHandle, shared_from_this()));
956 }
957
958 return chunks;

459inline std::shared_ptr<InterfaceDescriptor> System::GetOrAddFoundInterface(
460 PEAK_INTERFACE_DESCRIPTOR_HANDLE foundInterface)
461{
462 std::lock_guard<std::mutex> lock(m_interfacesMutex);
463
464 const auto interfaceKey = QueryStringFromCInterfaceFunction([&](char* key, size_t* keySize) {
465 return PEAK_C_ABI_PREFIX PEAK_InterfaceDescriptor_GetKey(foundInterface, key, keySize);
466 });
467 auto keyInterfacePairIterator = m_interfacesByKey.find(interfaceKey);
468 if (keyInterfacePairIterator == m_interfacesByKey.end())
469 {
470 auto interfaceDescriptor = std::static_pointer_cast<InterfaceDescriptor>(
471 std::make_shared<ClassCreator<InterfaceDescriptor>>(
472 foundInterface, std::weak_ptr<System>{ shared_from_this() }));
473 m_interfaces.emplace_back(interfaceDescriptor);
474 m_interfacesByKey.emplace(interfaceKey, interfaceDescriptor);
475 m_interfacesKeyById.emplace(interfaceDescriptor->ID(), interfaceKey);
476
477 return interfaceDescriptor;
478 }
479
480 return keyInterfacePairIterator->second;
481}

504inline void System::InitializeUpdateMechanismIfNecessary()
505{
506 std::call_once(m_updateMechanismInitializedFlag, [&] {
507 (void)RegisterInterfaceFoundCallback([](const std::shared_ptr<InterfaceDescriptor>&) {
508 // Registering an empty callback is enough since InterfaceFoundCallbackCWrapper() handles everything else.
509 });
510 (void)RegisterInterfaceLostCallback([&](const std::string& lostInterfaceId) {
511 std::lock_guard<std::mutex> lock(m_interfacesMutex);
512 const auto lostInterfaceKey = m_interfacesKeyById.at(lostInterfaceId);
513 m_interfaces.erase(std::remove_if(std::begin(m_interfaces), std::end(m_interfaces),
514 [lostInterfaceKey](const std::shared_ptr<InterfaceDescriptor>& interface_) {
515 return interface_->Key() == lostInterfaceKey;
516 }),
517 std::end(m_interfaces));
518 m_interfacesByKey.erase(lostInterfaceKey);
519 m_interfacesKeyById.erase(lostInterfaceId);
520 });
521 });
522}

45inline UsableRoiChunkInfo ExtractUsableRoiChunkInfo(const std::shared_ptr<peak::core::BufferChunk>& chunk)
46{
47 if (chunk->ID() != usableRoiChunkId)

68inline ExtractedPayload UpdatePayloadInfoWithChunks(const peak::core::detail::ExtractedPayload& payload,
69 const std::vector<std::shared_ptr<peak::core::BufferChunk>>& chunks)
70{
71 const auto imageChunk = FindBufferChunkById(chunks, imageChunkId);
72 if (imageChunk == nullptr)
73 {
74 throw core::InternalErrorException("Buffer has no image chunk.");
75 }
76
77 const auto imageSizeWithPaddingWithoutChunks = imageChunk->Size();
78
79 const auto usableRoiChunk = FindBufferChunkById(chunks, usableRoiChunkId);
80 if (usableRoiChunk == nullptr)
81 {
82 return payload;
83 }
84
85 ExtractedPayload result{ payload };
86
87 const auto chunkInfo = ExtractUsableRoiChunkInfo(usableRoiChunk);
88
89 result.size = peak::common::Size{ chunkInfo.usableWidth, chunkInfo.usableHeight };
90
91 auto buffer = imageChunk->ParentBuffer();
92 if (result.bytesPerLine != imageSizeWithPaddingWithoutChunks / buffer->Height())
93 {
94 throw core::InternalErrorException("Unexpected pitch!");
95 }
96
97 const auto bytesToSkip = result.bytesPerLine * chunkInfo.offsetY;
98 result.dataPtr = static_cast<void*>(static_cast<uint8_t*>(result.dataPtr) + bytesToSkip);
99
100 const peak::common::PixelFormatInfo pixelFormatInfo(result.pixelFormat);
101 result.lineStartOffset = pixelFormatInfo.GetSizeInBytes({ chunkInfo.offsetX, 1 });
102 result.dataSize = pixelFormatInfo.GetSizeInBytes(
103 { static_cast<uint32_t>(buffer->Width()), chunkInfo.usableHeight });
104 if (result.dataSize > imageSizeWithPaddingWithoutChunks)
105 {
106 throw core::InternalErrorException("Chunk size is not large enough to fit the expected image size!");
107 }
108
109 return result;
110}

112inline ExtractedPayload ExtractPayloadTypeImage(const std::shared_ptr<peak::core::Buffer>& buffer)
113{
114 assert(buffer->PixelFormatNamespace() != core::PixelFormatNamespace::IIDC
115 && buffer->PixelFormatNamespace() != core::PixelFormatNamespace::Custom);
116
117 ExtractedPayload result{};
118
119 result.size = { static_cast<uint32_t>(buffer->Width()), static_cast<uint32_t>(buffer->Height()) };
120 result.pixelFormat = static_cast<peak::common::PixelFormat>(buffer->PixelFormat());
121 result.dataPtr = static_cast<uint8_t*>(buffer->BasePtr()) + buffer->ImageOffset();
122
123 const auto pixelFormatInfo = peak::common::PixelFormatInfo(result.pixelFormat);
124
125 result.bytesPerLine = pixelFormatInfo.GetSizeInBytes({ static_cast<uint32_t>(buffer->Width()), 1 });
126 result.lineStartOffset = 0;
127 result.dataSize = pixelFormatInfo.GetSizeInBytes(
128 { static_cast<uint32_t>(buffer->Width()), static_cast<uint32_t>(buffer->Height()) });
129
130 const auto imageSizeWithPadding = buffer->Size() - buffer->ImageOffset();
131 if (result.dataSize > imageSizeWithPadding)
132 {
133 throw core::InternalErrorException("Buffer is not large enough to fit the expected image size!");
134 }
135
136 if (buffer->HasChunks())
137 {
138 result = UpdatePayloadInfoWithChunks(result, buffer->Chunks());
139 }
140
141 return result;
142}

144inline ExtractedPayload ExtractPayloadTypeChunk(const std::shared_ptr<peak::core::Buffer>& buffer)
145{
146 const auto remoteNodeMap = buffer->ParentDataStream()->ParentDevice()->RemoteDevice()->NodeMaps().at(0);
147 remoteNodeMap->UpdateChunkNodes(buffer);
148
149 const auto chunkWidth = remoteNodeMap->TryFindNode<core::nodes::IntegerNode>("ChunkWidth");
150 if (chunkWidth == nullptr || chunkWidth->AccessStatus() == core::nodes::NodeAccessStatus::NotAvailable)
151 {
152 throw core::InvalidCastException("Buffer has no ChunkWidth.");
153 }
154
155 const auto chunkHeight = remoteNodeMap->TryFindNode<core::nodes::IntegerNode>("ChunkHeight");
156 if (chunkHeight == nullptr || chunkHeight->AccessStatus() == core::nodes::NodeAccessStatus::NotAvailable)
157 {
158 throw core::InvalidCastException("Buffer has no ChunkHeight.");
159 }
160
161 const peak::common::Size size(
162 static_cast<uint32_t>(chunkWidth->Value()), static_cast<uint32_t>(chunkHeight->Value()));
163
164 const auto chunkPixelFormat = remoteNodeMap->TryFindNode<core::nodes::EnumerationNode>("ChunkPixelFormat");
165 if (chunkPixelFormat == nullptr || chunkPixelFormat->AccessStatus() == core::nodes::NodeAccessStatus::NotAvailable)
166 {
167 throw core::InvalidCastException("Buffer has no ChunkPixelFormat.");
168 }
169
170 const auto pixelFormat = static_cast<peak::common::PixelFormat>(
171 chunkPixelFormat->CurrentEntry()->NumericValue());
172
173 // assume first chunks is image data
174 const auto imageChunk = FindBufferChunkById(buffer->Chunks(), imageChunkId);
175 if (imageChunk == nullptr)
176 {
177 throw core::InternalErrorException("Buffer has no image chunk.");
178 }
179
180 const auto sizeInBytes = imageChunk->Size();
181
182 const auto pixelFormatInfo = peak::common::PixelFormatInfo(pixelFormat);
183 const auto expectedDataSize = pixelFormatInfo.GetSizeInBytes(size);
184 if (sizeInBytes < expectedDataSize)
185 {
186 std::stringstream msg;
187 msg << "The buffer's first chunk's size (" << imageChunk->Size() << ") is smaller than the expected data size ("
188 << expectedDataSize << ").";
189 throw core::InvalidCastException(msg.str());
190 }
191
192 return { size, pixelFormat, imageChunk->BasePtr(), sizeInBytes,
193 pixelFormatInfo.GetSizeInBytes(peak::common::Size(size.GetWidth(), 1)), 0 };
194}

196inline ExtractedPayload ExtractPayloadFromBufferPart(const std::shared_ptr<peak::core::BufferPart>& part)
197{
198 if (part == nullptr)
199 {
200 throw core::InvalidArgumentException("The given part is a nullptr!");
201 }
202
203 const auto partType = part->Type();
204
205 if (partType != BufferPartType::Image2D && partType != BufferPartType::ConfidenceMap
206 && partType != BufferPartType::Image3D)
207 {
208 throw core::InvalidCastException("BufferPart has no image data.");
209 }
210
211 assert(part->FormatNamespace() != static_cast<uint64_t>(core::PixelFormatNamespace::IIDC)
212 && part->FormatNamespace() != static_cast<uint64_t>(core::PixelFormatNamespace::Custom));
213
214 const auto pixelFormat = static_cast<peak::common::PixelFormat>(part->Format());
215 const auto bytesPerLine = peak::common::PixelFormatInfo(pixelFormat)
216 .GetSizeInBytes({ static_cast<uint32_t>(part->Width()), 1 });
217 return { peak::common::Size{ static_cast<uint32_t>(part->Width()), static_cast<uint32_t>(part->Height()) },
218 pixelFormat, part->BasePtr(), part->Size(), bytesPerLine, 0 };
219}

221inline ExtractedPayload ExtractPayloadFromBuffer(const std::shared_ptr<peak::core::Buffer>& buffer)
222{
223 if (buffer == nullptr)
224 {
225 throw core::InvalidArgumentException("The given buffer is a nullptr!");
226 }
227
228 if (buffer->HasImage())
229 {
230 return ExtractPayloadTypeImage(buffer);
231 }
232
233 if (buffer->PayloadType() == core::BufferPayloadType::Chunk)
234 {
235 return ExtractPayloadTypeChunk(buffer);
236 }
237
238 throw core::InvalidCastException("Buffer has no image data and no chunks.");
239}

241inline std::shared_ptr<peak::common::Metadata> ExtractMetadataFromBuffer(
242 const std::shared_ptr<peak::core::Buffer>& buffer, MetadataExtractionMode mode)
243{
244 if (buffer == nullptr)
245 {
246 throw core::InvalidArgumentException("The given buffer is a nullptr!");
247 }
248
249 auto metadata = std::make_shared<peak::common::Metadata>();
250
251 // NOTE: BUFFER_INFO_FRAMEID is a mandatory GenTL feature. It MUST be implemented.
252 metadata->SetValueByKey<peak::common::MetadataKey::DeviceFrameID>(buffer->FrameID());
253
254 // NOTE: We need to try, as the timestamp might not be supported by the TL.
255 try
256 {
257 metadata->SetValueByKey<peak::common::MetadataKey::DeviceTimestamp>(buffer->Timestamp_ns());
258 }
259 catch (const peak::core::Exception&)
260 {}
261
262 // NOTE: The system timestamp is optional and may not be supported by the TL or device.
263 try
264 {
265 metadata->SetValueByKey<peak::common::MetadataKey::SystemTimestamp>(buffer->SystemTimestamp_ns());
266 }
267 catch (const peak::core::Exception&)
268 {}
269
270 if (mode == MetadataExtractionMode::Minimal)
271 {
272 return metadata;
273 }
274
275 // MetadataExtractionMode::Extended
276
277 const auto remoteNodeMap = buffer->ParentDataStream()->ParentDevice()->RemoteDevice()->NodeMaps().at(0);
278 const auto chunkExposureTimeNode = remoteNodeMap->TryFindNode<nodes::FloatNode>("ChunkExposureTime");
279 const auto chunkGainSelectorNode = remoteNodeMap->TryFindNode<nodes::EnumerationNode>("ChunkGainSelector");
280 const auto chunkGainNode = remoteNodeMap->TryFindNode<nodes::FloatNode>("ChunkGain");
281
282 const auto lock = remoteNodeMap->Lock();
283 remoteNodeMap->UpdateChunkNodes(buffer);
284
285 if (chunkExposureTimeNode != nullptr && chunkExposureTimeNode->IsReadable())
286 {
287 metadata->SetValueByKey<peak::common::MetadataKey::DeviceExposureTime>(chunkExposureTimeNode->Value());
288 }
289
290 if (chunkGainSelectorNode != nullptr && chunkGainSelectorNode->AccessStatus() == nodes::NodeAccessStatus::ReadWrite
291 && chunkGainNode != nullptr)
292 {
293 const auto availableEntries = chunkGainSelectorNode->AvailableEntries();
294
295 bool found = false;
296 auto readChunkGain = [&](const std::string& entryName) {
297 const auto entry = chunkGainSelectorNode->TryFindEntry(entryName);
298 if (entry == nullptr || !entry->IsAvailable())
299 {
300 return 1.0;
301 }
302
303 chunkGainSelectorNode->SetCurrentEntry(entry);
304 if (!chunkGainNode->IsReadable())
305 {
306 return 1.0;
307 }
308 found = true;
309 return chunkGainNode->Value();
310 };
311
312 const auto gainAll = readChunkGain("All");
313 const auto gainAnalogAll = readChunkGain("AnalogAll");
314 const auto gainDigitalAll = readChunkGain("DigitalAll");
315
316 if (found)
317 {
318 metadata->SetValueByKey<peak::common::MetadataKey::DeviceGain>(
319 gainAll * gainAnalogAll * gainDigitalAll);
320 }
321 }
322
323 return metadata;
324}

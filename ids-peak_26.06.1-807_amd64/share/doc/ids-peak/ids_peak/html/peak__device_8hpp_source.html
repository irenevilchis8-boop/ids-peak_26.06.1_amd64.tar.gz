306inline void Device::Initialize() const
307{
308 EventSupportingModule::Initialize(); // init data of parent classes (i.e. Module) as well
309
310 // init data streams
311 auto numDataStreams = QueryNumericFromCInterfaceFunction<size_t>([&](size_t* _numDataStreams) {
312 return PEAK_C_ABI_PREFIX PEAK_Device_GetNumDataStreams(m_backendHandle, _numDataStreams);
313 });
314
315 std::vector<std::shared_ptr<DataStreamDescriptor>> dataStreams;
316 for (size_t x = 0; x < numDataStreams; ++x)
317 {
318 auto dataStreamDescriptorHandle = QueryNumericFromCInterfaceFunction<PEAK_DATA_STREAM_DESCRIPTOR_HANDLE>(
319 [&](PEAK_DATA_STREAM_DESCRIPTOR_HANDLE* _dataStreamDescriptorHandle) {
320 return PEAK_C_ABI_PREFIX PEAK_Device_GetDataStream(
321 m_backendHandle, x, _dataStreamDescriptorHandle);
322 });
323
324 dataStreams.emplace_back(std::make_shared<ClassCreator<DataStreamDescriptor>>(
325 dataStreamDescriptorHandle, std::const_pointer_cast<Device>(shared_from_this())));
326 }
327
328 m_dataStreams = dataStreams;
329
330 // init remote device
331 auto remoteDeviceHandle = QueryNumericFromCInterfaceFunction<PEAK_REMOTE_DEVICE_HANDLE>(
332 [&](PEAK_REMOTE_DEVICE_HANDLE* _remoteDeviceHandle) {
333 return PEAK_C_ABI_PREFIX PEAK_Device_GetRemoteDevice(m_backendHandle, _remoteDeviceHandle);
334 });
335
336 m_remoteDevice = std::make_shared<ClassCreator<peak::core::RemoteDevice>>(
337 remoteDeviceHandle, std::const_pointer_cast<Device>(shared_from_this()));
338}

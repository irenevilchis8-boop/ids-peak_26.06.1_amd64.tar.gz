309 ExecuteAndMapReturnCodes(

311
312 for (size_t ch = 0; ch < numChannels; ++ch)
313 {
314 size_t valueListSize = 0;
315 ExecuteAndMapReturnCodes([&] {
316 return PEAK_IPL_C_ABI_PREFIX PEAK_IPL_PixelLine_GetValuesForChannel(
317 m_backendHandle, ch, nullptr, &valueListSize);
318 });
319 std::vector<uint32_t> valueList(valueListSize);
320 ExecuteAndMapReturnCodes([&] {
321 return PEAK_IPL_C_ABI_PREFIX PEAK_IPL_PixelLine_GetValuesForChannel(
322 m_backendHandle, ch, valueList.data(), &valueListSize);
323 });
324
325 channels.emplace_back(Channel{ std::move(valueList) });
326 }
327
328 return channels;

226inline std::vector<Histogram::HistogramChannel> Histogram::Channels() const
227{
228 std::vector<HistogramChannel> channels;
229
230 size_t numChannels = 0;
231 ExecuteAndMapReturnCodes(
232 [&] { return PEAK_IPL_C_ABI_PREFIX PEAK_IPL_Histogram_GetNumChannels(m_backendHandle, &numChannels); });
233
234 for (size_t ch = 0; ch < numChannels; ++ch)
235 {
236 uint64_t pixelSum = 0;
237 ExecuteAndMapReturnCodes([&] {
238 return PEAK_IPL_C_ABI_PREFIX PEAK_IPL_Histogram_GetPixelSumForChannel(m_backendHandle, ch, &pixelSum);
239 });
240
241 uint64_t pixelCount = 0;
242 ExecuteAndMapReturnCodes([&] {
243 return PEAK_IPL_C_ABI_PREFIX PEAK_IPL_Histogram_GetPixelCountForChannel(m_backendHandle, ch, &pixelCount);
244 });
245
246 size_t binListSize = 0;
247 ExecuteAndMapReturnCodes([&] {
248 return PEAK_IPL_C_ABI_PREFIX PEAK_IPL_Histogram_GetBinsForChannel(m_backendHandle, ch, nullptr, &binListSize);
249 });
250 std::vector<uint64_t> binList(binListSize);
251 ExecuteAndMapReturnCodes([&] {
252 return PEAK_IPL_C_ABI_PREFIX PEAK_IPL_Histogram_GetBinsForChannel(
253 m_backendHandle, ch, binList.data(), &binListSize);
254 });
255
256 channels.emplace_back(HistogramChannel{ pixelSum, pixelCount, std::move(binList) });
257 }
258
259 return channels;
260}

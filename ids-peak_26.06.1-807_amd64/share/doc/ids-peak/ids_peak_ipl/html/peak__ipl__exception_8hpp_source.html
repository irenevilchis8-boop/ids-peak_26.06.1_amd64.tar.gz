288template <class CallableType>
289void ExecuteAndMapReturnCodes(const CallableType& callableObject)
290{
291 if (callableObject() != PEAK_IPL_RETURN_CODE_SUCCESS)
292 {
293 PEAK_IPL_RETURN_CODE lastErrorCode = PEAK_IPL_RETURN_CODE_SUCCESS;
294 size_t lastErrorDescriptionSize = 0;
295 if (PEAK_IPL_C_ABI_PREFIX PEAK_IPL_Library_GetLastError(&lastErrorCode, nullptr, &lastErrorDescriptionSize)
296 != PEAK_IPL_RETURN_CODE_SUCCESS)
297 {
298 throw Exception("Could not query the last error!");
299 }
300 std::vector<char> lastErrorDescription(lastErrorDescriptionSize);
301 if (PEAK_IPL_C_ABI_PREFIX PEAK_IPL_Library_GetLastError(
302 &lastErrorCode, lastErrorDescription.data(), &lastErrorDescriptionSize)
303 != PEAK_IPL_RETURN_CODE_SUCCESS)
304 {
305 throw Exception("Could not query the last error!");
306 }
307
308 std::stringstream errorText;
309 errorText << "[Error-Code: " << lastErrorCode << " ("
310 << StringFromPEAK_IPL_RETURN_CODE(static_cast<PEAK_IPL_RETURN_CODE_t>(lastErrorCode))
311 << ") | Error-Description: " << lastErrorDescription.data() << "]";
312
313 switch (lastErrorCode)
314 {
315 case PEAK_IPL_RETURN_CODE_OUT_OF_RANGE:
316 throw OutOfRangeException(errorText.str().c_str());
317 case PEAK_IPL_RETURN_CODE_INVALID_ARGUMENT:
318 throw InvalidArgumentException(errorText.str().c_str());
319 case PEAK_IPL_RETURN_CODE_IMAGE_FORMAT_NOT_SUPPORTED:
320 throw ImageFormatNotSupportedException(errorText.str().c_str());
321 case PEAK_IPL_RETURN_CODE_FORMAT_INTERPRETATION_ERROR:
322 throw ImageFormatInterpretationException(errorText.str().c_str());
323 case PEAK_IPL_RETURN_CODE_IO_ERROR:
324 throw IOException(errorText.str().c_str());
325 case PEAK_IPL_RETURN_CODE_BUFFER_TOO_SMALL:
326 throw BufferTooSmallException(errorText.str().c_str());
327 case PEAK_IPL_RETURN_CODE_NOT_PERMITTED:
328 throw NotPermittedException(errorText.str().c_str());
329 case PEAK_IPL_RETURN_CODE_BUSY:
330 throw BusyException(errorText.str().c_str());
331 case PEAK_IPL_RETURN_CODE_INVALID_HANDLE:
332 throw InvalidHandleException(errorText.str().c_str());
333 case PEAK_IPL_RETURN_CODE_TIMEOUT:
334 throw TimeoutException(errorText.str().c_str());
335 default:
336 throw Exception(errorText.str().c_str());
337 }
338 }
339}

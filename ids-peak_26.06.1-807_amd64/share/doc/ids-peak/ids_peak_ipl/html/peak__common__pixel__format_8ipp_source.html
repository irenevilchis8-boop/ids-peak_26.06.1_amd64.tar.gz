60template <PixelFormat pixelFormat>
61struct PixelFormatInfo
62{
63 using traits = PixelFormatTraits<pixelFormat>;
64 using underlying_type = typename traits::underlying_type;
65 static constexpr bool is_floating_point = std::is_floating_point<underlying_type>::value;
66
67 static constexpr size_t storageBitsPerPixel = traits::storageBitsPerPixel;
68 static constexpr size_t allocatedBitsPerPixel = traits::allocatedBitsPerPixel;
69
70 static constexpr bool isYUV = storageBitsPerPixel / traits::Channels().size() < 8;
71
72 static constexpr size_t storageBitsPerChannel = isYUV ? storageBitsPerPixel : storageBitsPerPixel / traits::Channels().size();
73
74 using unpacked_underlying_type = typename PixelFormatTraits<traits::unpackedFormat>::underlying_type;
75
76 static constexpr unpacked_underlying_type minimumValuePerChannel =
